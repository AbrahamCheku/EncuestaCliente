<script>

const form=document.getElementById('brandForm');
const steps=document.querySelectorAll('.form-step');
const bar=document.getElementById('progressBar');
const label=document.getElementById('stepLabel');
const percent=document.getElementById('progressPercent');
const prev=document.getElementById('prevBtn');
const next=document.getElementById('nextBtn');
let current=0;

function update(){
  steps.forEach((s,i)=>s.classList.toggle('active',i===current));
  const pct=Math.round(((current+1)/steps.length)*100);
  bar.style.width=pct+'%';
  label.textContent=`SECCIÓN ${String(current+1).padStart(2,'0')} / ${steps.length}`;
  percent.textContent=pct+'%';
  prev.style.visibility=current===0?'hidden':'visible';
  next.style.display=current===steps.length-1?'none':'block';
  window.scrollTo({top:document.getElementById('brief').offsetTop-30,behavior:'smooth'});
}
function validateStep(){
  const required=steps[current].querySelectorAll('[required]');
  for(const el of required){
    if(!el.value.trim()){
      el.focus();
      el.style.borderColor='var(--red)';
      setTimeout(()=>el.style.borderColor='',1500);
      return false;
    }
  }
  return true;
}
function nextStep(){if(validateStep() && current<steps.length-1){current++;update();}}
function prevStep(){if(current>0){current--;update();}}

function getData(){
  const data={};
  const fd=new FormData(form);
  for(const [key,value] of fd.entries()){
    if(data[key]) data[key]+=', '+value;
    else data[key]=value;
  }
  return data;
}

function buildMessage(){
  const d=getData();
  const section=(title, rows)=>{
    const valid=rows.filter(x=>x[1]&&String(x[1]).trim());
    if(!valid.length) return '';
    return `*${title}*\n`+valid.map(x=>`• *${x[0]}:* ${String(x[1]).trim()}`).join('\n');
  };

  const parts=[
    `*BRIEF DE IDENTIDAD DE MARCA*`,
    `━━━━━━━━━━━━━━━━━━━━`,
    `Hola, te comparto la información para desarrollar mi identidad de marca.`,
    '',
    section('01 · LA EMPRESA',[
      ['Empresa',d.empresa],['Nombre',d.nombre],['Correo',d.email],
      ['Actividad',d.giro],['Productos / servicios',d.productos],['Mercado / zona',d.mercado]
    ]),
    section('02 · ESENCIA',[
      ['Propósito',d.proposito],['Diferenciador',d.diferenciador],
      ['Cliente ideal',d.publico],['Valores',d.valores]
    ]),
    section('03 · PERSONALIDAD',[
      ['Personalidad',d.personalidad],['Qué evitar',d.evitar],
      ['Marca como persona',d.persona]
    ]),
    section('04 · DIRECCIÓN VISUAL',[
      ['Tipo de logo',d.tipo_logo],['Colores deseados',d.colores],
      ['Colores a evitar',d.colores_no],['Referencias',d.referencias],
      ['Símbolos / conceptos',d.simbolo]
    ]),
    section('05 · APLICACIONES',[
      ['Aplicaciones',d.aplicaciones],['Necesito',d.manual],
      ['Comentarios',d.comentarios]
    ]),
    `━━━━━━━━━━━━━━━━━━━━`,
    `*Gracias. Quedo pendiente para continuar con el proceso creativo.*`
  ];

  return parts.filter(Boolean).join('\n\n');
}





function submitBrief(e){
  e.preventDefault();
  if(current!==steps.length-1 || !validateStep()) return;

  const d=getData();
  const message=buildMessage();
  const wa='529987684125';

  const status=document.getElementById('formStatus');
  status.style.display='block';
  status.innerHTML=`
    <div class="whatsapp-confirm">
      <div class="wa-label">05 / FINAL</div>
      <h3>ABRIENDO<br><span style="color:var(--red)">WHATSAPP.</span></h3>
      <p>Tu brief está listo. WhatsApp se abrirá con todas tus respuestas ordenadas y preparadas para enviar.</p>
    </div>`;

  const url='https://wa.me/'+wa+'?text='+encodeURIComponent(message);
  setTimeout(()=>{ window.location.href=url; },250);
}

const manualSelect=document.getElementById('manualSelect');
if(manualSelect){
  const manualTrigger=manualSelect.querySelector('.custom-select-trigger');
  const manualSelected=document.getElementById('manualSelected');
  const manualValue=document.getElementById('manualValue');
  const manualOptions=manualSelect.querySelectorAll('.custom-select-menu button');

  manualTrigger.addEventListener('click',()=>{
    const open=manualSelect.classList.toggle('open');
    manualTrigger.setAttribute('aria-expanded',open?'true':'false');
  });

  manualOptions.forEach(option=>{
    option.addEventListener('click',()=>{
      manualValue.value=option.dataset.value;
      manualSelected.textContent=option.textContent;
      manualOptions.forEach(o=>o.classList.remove('selected'));
      option.classList.add('selected');
      manualSelect.classList.remove('open');
      manualTrigger.setAttribute('aria-expanded','false');
    });
  });

  document.addEventListener('click',(event)=>{
    if(!manualSelect.contains(event.target)){
      manualSelect.classList.remove('open');
      manualTrigger.setAttribute('aria-expanded','false');
    }
  });
}

form.addEventListener('submit',submitBrief);
update();
